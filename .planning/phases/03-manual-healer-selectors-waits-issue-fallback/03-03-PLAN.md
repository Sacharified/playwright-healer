```typescript
import {
  FORBIDDEN_PATCHED_LINE_PATTERNS,
  ASSERTION_WEAKENING_PAIRS,
  TEST_PATH_ALLOWLIST,
} from './forbidden-patterns.js';

export interface LintFinding {
  pattern: string;
  filePath: string;
  hunkLine: number;
  excerpt: string;
}

export function lintDiff(unifiedDiff: string): LintFinding[] {
  const findings: LintFinding[] = [];
  // 1. Walk lines; track current `+++ b/<path>` filePath as we encounter file headers.
  //    Track current hunk's added (+) lines and removed (-) lines (paired by hunk).
  // 2. For each `+` line (excluding `+++ ` headers): test against each
  //    FORBIDDEN_PATCHED_LINE_PATTERNS entry; push a finding on match.
  // 3. For each ASSERTION_WEAKENING_PAIRS pair: if ANY `-` line in the current
  //    hunk matches `from` AND ANY `+` line in the same hunk matches `to`,
  //    push a finding tagged with the pair name (e.g., 'toBe-to-toBeTruthy').
  // 4. For each unique filePath that appears in a `+++ b/<path>` header AND
  //    has any `+` line (i.e., the file is modified or added — not pure delete):
  //    if NONE of TEST_PATH_ALLOWLIST regexes match `<path>`, push a finding
  //    with pattern: 'out-of-test-dir', filePath, hunkLine: 0.
  return findings;
}
```

The implementation should:
- Parse `+++ b/<path>` headers (the standard unified-diff "after" path; ignore `--- a/<path>` "before" path)
- Treat `/dev/null` after-paths as deletions (no allowlist check needed for pure deletions)
- Track per-hunk state for the assertion-weakening check (a `-.toBe(...)` followed by `+.toBeTruthy(...)` within the same `@@ ... @@` hunk = weakening)
