```typescript
// src/index.ts
//
// playwright-healer — composite GitHub Action entry point.
//
// Startup ordering (D-07) is AUTHORITATIVE:
//   1. getInput() the three secret inputs
//   2. core.setSecret() each one — registers with runner mask BEFORE any log line
//   3. getInput() the non-secret inputs
//   4. Zod validation (fail-fast with field-naming error on invalid input)
//   5. switch-dispatch on mode: dry-run is self-contained; ingest/heal dynamically
//      import their stub modules (throw in Phase 1 per D-09)
//
// IMPORTANT: do NOT log anything — core.info, console.log, throw-with-input-values —
// before step 2 completes. Any log line before setSecret leaks the value into the
// Actions log.

import * as core from '@actions/core';
import { getInputSchema, type Config } from './shared/config.js';

async function main(): Promise<void> {
  // ── Phase A: SECRET MASKING (D-07 — must be first, before any log line) ──
  const anthropicApiKey = core.getInput('anthropic-api-key', { required: true });
  const healerToken     = core.getInput('healer-token',      { required: true });
  const githubToken     = core.getInput('github-token',      { required: true });

  core.setSecret(anthropicApiKey);
  core.setSecret(healerToken);
  core.setSecret(githubToken);

  // ── Phase B: INPUT COLLECTION ──
  const rawInputs = {
    mode:           core.getInput('mode',           { required: true }),
    setupCommand:   core.getInput('setup-command'),
    startCommand:   core.getInput('start-command'),
    testCommand:    core.getInput('test-command'),
    baseUrl:        core.getInput('base-url'),
    anthropicApiKey,
    healerToken,
    githubToken,
  };

  // ── Phase C: VALIDATION (Zod; fail-fast; field-naming error per SC#4) ──
  const parsed = getInputSchema().safeParse(rawInputs);
  if (!parsed.success) {
    const msg = parsed.error.issues
      .map((i) => `${i.path.join('.')}: ${i.message}`)
      .join('; ');
    core.setFailed(`Invalid inputs: ${msg}`);
    return; // setFailed sets exit code 1; do not call process.exit
  }
  const config: Config = parsed.data;

  // ── Phase D: DISPATCH ──
  switch (config.mode) {
    case 'dry-run':
      await runDryRun(config);
      return;
    case 'ingest': {
      const m = await import('./ingest/index.js');
      await m.run(config); // throws in Phase 1 per D-09
      return;
    }
    case 'heal': {
      const m = await import('./healer/index.js');
      await m.run(config); // throws in Phase 1 per D-09
      return;
    }
  }
}

/**
 * Dry-run: write a redacted config summary to $GITHUB_STEP_SUMMARY and exit 0.
 * Per D-05, the three secrets are NEVER emitted (omitted rather than replaced
 * with ***, because the runner mask would replace them anyway — belt and
 * suspenders). This is the permanent dry-run contract: exit 0 + no side effects.
 */
async function runDryRun(config: Config): Promise<void> {
  const rows: Array<[string, string]> = [
    ['mode',          config.mode],
    ['setup-command', config.setupCommand || '(empty)'],
    ['start-command', config.startCommand || '(empty)'],
    ['test-command',  config.testCommand  || '(empty)'],
    ['base-url',      config.baseUrl      || '(empty)'],
    // Secrets intentionally omitted from dry-run output (defense in depth on top of setSecret masking).
    ['anthropic-api-key', '(set — redacted)'],
    ['healer-token',      '(set — redacted)'],
    ['github-token',      '(set — redacted)'],
  ];

  let md = '# playwright-healer — dry-run summary\n\n';
  md += '| Input | Value |\n| --- | --- |\n';
  for (const [k, v] of rows) {
    md += `| \`${k}\` | ${v} |\n`;
  }
  md += '\n_No side effects were performed. Exit code: 0._\n';

  await core.summary.addRaw(md).write();
}

main().catch((err) => {
  core.setFailed(err instanceof Error ? err.message : String(err));
});
```

**Critical non-obvious requirements** (PATTERNS items 1-6):

  1. **Startup order is AUTHORITATIVE (D-07)** — getInput three secrets → setSecret three secrets → getInput everything else → Zod validate → dispatch. Breaking this order (e.g., `core.info('starting')` before `setSecret`) leaks secrets into the log. Plan 06 self-test job `self-test-masking` + `verify-log-mask` will verify this mechanically.

  2. **Secret `getInput` happens BEFORE `setSecret`** — correct. `setSecret(v)` registers the already-retrieved value `v` with the runner's mask filter; calling `setSecret` on an un-retrieved value would be nonsensical.

  3. **`core.setFailed(msg); return;`** — NOT `process.exit(1)`. `setFailed` sets the exit code to 1 AND records the failure in the step status; `process.exit` bypasses async cleanup handlers. Grep will verify absence of `process.exit` in this file.

  4. **Zod error formatting MUST name the field** (SC#4). `issue.path.join('.')` produces `mode`, `anthropicApiKey`, etc. Those are the camelCase schema field names. The message includes both the path AND the Zod-provided `message` string — so for `mode: 'banana'`, the error is like `Invalid inputs: mode: mode must be one of: ingest, heal, dry-run`. Good enough for SC#4 (error names the field).

  5. **Dynamic `import('./ingest/index.js')`** — NOT a static top-level import. Keeps the stubs from being evaluated during `dry-run` (important for Phase 2/3 when stubs become heavyweight SDK/MCP code). The `.js` extension (not `.ts`) matches `moduleResolution: bundler` in tsconfig — tsx resolves `.js` to the `.ts` source at runtime.

  6. **Dry-run summary writes to `$GITHUB_STEP_SUMMARY`** via `core.summary.addRaw(...).write()`. Secrets are OMITTED from the table (rendered as `(set — redacted)`). Even if accidentally logged elsewhere, the runner mask replaces them — defense-in-depth layered on top of the omission.

**What NOT to do:**
  - NO `console.log`, `core.info`, `core.debug`, or any other logging BEFORE all three `setSecret` calls complete. Any log call before masking is registered leaks the secret.
  - NO API calls, git operations, dispatch events, or any side effect before Zod validation passes — D-07 is explicit.
  - NO `process.exit(1)` anywhere — use `core.setFailed(msg); return;` and let the runtime exit via the main().catch() handler.
  - NO inline string literal for `allowedTools`, `mcp__playwright__*`, `pull_request_target`, or any security-contract value. These are only in `src/shared/security-contract.ts` (Phase 1 dispatcher does not use them; Phase 3 will IMPORT them from there).
  - NO `import fetch from 'node-fetch'` or any HTTP client — SEC-07 grep in Plan 05 will fail if any of `fetch(`, `http.request(`, `https.request(`, `axios`, `got(`, `node-fetch`, `undici` appear in this file.
  - NO static import of `./ingest/index.js` or `./healer/index.js` — use dynamic import inside the switch case so they only evaluate when their mode is selected.

**`@actions/core` summary API note:** `core.summary.addRaw(md).write()` returns a `Promise<Summary>`. Must `await` it so the write completes before the process exits. If `$GITHUB_STEP_SUMMARY` env var is unset (e.g., running locally outside GitHub Actions), `write()` silently no-ops — the function does not throw. This is fine for Phase 1 testing.

| Dispatcher → stub modules | Stubs throw; main().catch() routes to core.setFailed. No secret values reach the stubs (only Config struct). |
