```typescript
// src/ingest/index.ts
import * as core from '@actions/core';
import * as github from '@actions/github';
import { create as createGlob } from '@actions/glob';
import * as fs from 'fs';
import * as path from 'path';
import { createRequire } from 'module';
import { shouldSkipIngest } from '../shared/loop-guard.js';
import { loadYamlConfig, mergeConfigs, type Config } from '../shared/config.js';
import { parseReport } from './report-parser.js';
import {
  bootstrapOrGetWorktree,
  appendRecord,
  runGc,
  removeWorktree,
  todayPath,
} from '../shared/state-branch.js';
import type { NdjsonRecord } from '../shared/types.js';
import { evaluateThresholds } from './threshold-evaluator.js';
import { writeDetectionSummary } from './summary-writer.js';

// Read package.json version for healerVersion field (relative to this compiled file)
const require = createRequire(import.meta.url);
const { version: VERSION } = require('../../package.json') as { version: string };

export async function run(config: Config): Promise<void> {
  // ── Step 1: LOOP GUARD (SEC-05) ───────────────────────────────────────────
  // Must be FIRST — before any config read, file read, or git operation
  if (shouldSkipIngest()) {
    return;  // early exit; shouldSkipIngest() emits the core.info() message
  }

  // ── Step 2: CONFIG MERGE (CFG-06, CFG-07) ─────────────────────────────────
  // Load optional YAML config file and merge with action.yml inputs (action wins)
  const workspacePath = process.env.GITHUB_WORKSPACE ?? process.cwd();
  const yamlConfig = loadYamlConfig(workspacePath);
  // config object already validated by Zod in src/index.ts; yamlConfig values for
  // any field NOT in action.yml inputs can be used here for display/logging purposes.
  // The Zod-validated `config` is authoritative — yamlConfig is informational only here.

  // ── Step 3: REPORT PARSE (ING-01..04) ─────────────────────────────────────
  const globber = await createGlob(config.reportPath, { followSymbolicLinks: false });
  const reportFiles = await globber.glob();

  // Shard metadata from environment (ING-04)
  const shardIndex = parseInt(process.env.SHARD_INDEX ?? '0', 10) || null;
  const shardTotal = parseInt(process.env.SHARD_TOTAL ?? '0', 10) || null;

  let allEntries: NdjsonRecord['tests'] = [];
  let reportUnreadable = false;

  if (reportFiles.length === 0) {
    core.warning(`ING-01: No report files matched pattern "${config.reportPath}". Recording as "report-unreadable".`);
    reportUnreadable = true;
  } else {
    for (const reportFile of reportFiles) {
      let rawJson: unknown;
      try {
        rawJson = JSON.parse(fs.readFileSync(reportFile, 'utf8'));
      } catch (err) {
        core.warning(`ING-01: Could not read/parse ${reportFile}: ${err}. Skipping.`);
        continue;
      }
      const { entries, reportUnreadable: isUnreadable } = parseReport(rawJson);
      if (isUnreadable) {
        reportUnreadable = true;
      } else {
        allEntries = allEntries.concat(entries);
      }
    }
  }

  // ── Step 4: BUILD NDJSON RECORD ──────────────────────────────────────────
  const record: NdjsonRecord = {
    schemaVersion: 1,
    timestamp: new Date().toISOString(),
    runId: process.env.GITHUB_RUN_ID ?? 'local',
    commitSha: github.context.sha,
    branch: github.context.ref,
    healerVersion: VERSION,
    shardIndex,
    shardTotal,
    tests: reportUnreadable
      ? allEntries.map((e) => ({ ...e, outcome: 'report-unreadable' as const }))
      : allEntries,
  };

  // ── Step 5: STATE BRANCH APPEND (STA-01..05) ─────────────────────────────
  const remoteUrl = `${process.env.GITHUB_SERVER_URL ?? 'https://github.com'}/${process.env.GITHUB_REPOSITORY}.git`;
  const primaryCwd = process.env.GITHUB_WORKSPACE ?? process.cwd();

  let worktreePath: string | null = null;
  try {
    worktreePath = await bootstrapOrGetWorktree(remoteUrl, primaryCwd);
    await appendRecord(record, worktreePath);

    // ── Step 5b: GC (STA-05) ─────────────────────────────────────────────
    await runGc(config.retentionDays, worktreePath);

    // ── Step 6: THRESHOLD EVALUATION (DET-01..04) ─────────────────────────
    const windowRecords = readWindowRecords(worktreePath, config.flakeWindowDays);
    const detections = evaluateThresholds(windowRecords, config);

    // ── Step 7: STEP SUMMARY (DET-04 log-only) ────────────────────────────
    await writeDetectionSummary(detections);

  } finally {
    // Always clean up the worktree, even on error
    if (worktreePath) {
      await removeWorktree(worktreePath).catch((e) =>
        core.warning(`Worktree cleanup failed: ${e}`)
      );
    }
  }
}

/**
 * Reads NDJSON records from the rolling window date range.
 * Only reads files within [today - flakeWindowDays, today] — not the entire corpus.
 * Malformed NDJSON lines are skipped with a warning (Pitfall B resilience).
 */
function readWindowRecords(
  worktreePath: string,
  flakeWindowDays: number,
): NdjsonRecord[] {
  const records: NdjsonRecord[] = [];
  const today = new Date();

  for (let daysBack = 0; daysBack <= flakeWindowDays; daysBack++) {
    const d = new Date(today);
    d.setDate(d.getDate() - daysBack);
    const y = d.getUTCFullYear();
    const m = String(d.getUTCMonth() + 1).padStart(2, '0');
    const day = String(d.getUTCDate()).padStart(2, '0');
    const filePath = path.join(worktreePath, 'runs', String(y), m, `${day}.ndjson`);

    if (!fs.existsSync(filePath)) continue;

    const lines = fs.readFileSync(filePath, 'utf8').split('\n').filter(Boolean);
    for (const line of lines) {
      try {
        records.push(JSON.parse(line) as NdjsonRecord);
      } catch {
        core.warning(`State branch: malformed NDJSON line skipped in ${filePath}`);
      }
    }
  }

  return records;
}
```

**Notes:**
- The `run(config: Config): Promise<void>` return type changes from `Promise<never>` (Phase 01 stub) to `Promise<void>`.
- `finally` block ensures worktree cleanup on any exit path (success, error, timeout).

- `yamlConfig` is loaded and could be used to extend logging/debug output, but since the Zod-validated `config` already has all threshold values (either from action inputs or Zod defaults), the YAML file's only runtime effect is providing values that override Zod defaults when action inputs are empty strings. This is already handled in `mergeConfigs` + Zod coercion in `src/index.ts`. The `loadYamlConfig()` call here is for Phase 03 extension point.

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Untrusted GITHUB_SERVER_URL / GITHUB_REPOSITORY env vars → remoteUrl | These are set by the GitHub Actions runner; in a self-hosted runner context, they could be modified. The constructed URL is only used for git push to state branch — not for reading or executing. |
| Consumer's report file → report-parser | File content from GITHUB_WORKSPACE (could be attacker-controlled on fork PR — mitigated by loop guard exit before this point). |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-2-02c | Repudiation (pipeline invoked on bot commit) | `src/ingest/index.ts` — shouldSkipIngest() placement | mitigate | shouldSkipIngest() is the FIRST call in run(). No config read, no file I/O, no git operation occurs before it returns. A bot commit returns immediately with core.info() — no side effects. |
| T-2-03b | Tampering (report glob outside workspace) | `@actions/glob` create() — report-path glob resolution | mitigate | `@actions/glob` by default constrains glob results to the GITHUB_WORKSPACE directory. A path traversal pattern like `../../etc/passwd` returns no results (outside workspace). If zero files match, the record is written as 'report-unreadable' — non-fatal. |
| T-2-06c | Elevation of privilege (dispatch in ingest) | Full `src/` tree | mitigate | `grep -rn 'createWorkflowDispatch' src/` returns zero results. Phase 04 is the only authorized phase to add dispatch; this verify command enforces the boundary. |
