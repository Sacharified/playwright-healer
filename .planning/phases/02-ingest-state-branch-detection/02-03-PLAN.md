```typescript
// src/shared/state-branch.ts
import { getExecOutput } from '@actions/exec';
import * as core from '@actions/core';
import * as fs from 'fs';
import * as os from 'os';
import * as path from 'path';
import type { NdjsonRecord } from './types.js';

const STATE_BRANCH = 'playwright-healer-state';
const BOT_EMAIL = 'playwright-healer-bot@users.noreply.github.com';
const BOT_NAME = 'playwright-healer-bot';
const MAX_RETRIES = 5;

export function todayPath(): string { /* runs/YYYY/MM/DD.ndjson UTC */ }

export async function bootstrapOrGetWorktree(
  repoRemoteUrl: string,
  primaryCwd: string = process.cwd(),
): Promise<string> {
  const worktreePath = path.join(os.tmpdir(), `playwright-healer-state-${Date.now()}`);

  const lsRemote = await getExecOutput(
    'git', ['ls-remote', '--exit-code', 'origin', `refs/heads/${STATE_BRANCH}`],
    { cwd: primaryCwd, ignoreReturnCode: true }
  );

  if (lsRemote.exitCode === 0) {
    // Branch exists — use git worktree add
    await getExecOutput('git', [
      'fetch', 'origin', STATE_BRANCH
    ], { cwd: primaryCwd });
    await getExecOutput('git', [
      'worktree', 'add', '--no-checkout', worktreePath, `origin/${STATE_BRANCH}`
    ], { cwd: primaryCwd });
    await getExecOutput('git', ['checkout', STATE_BRANCH], { cwd: worktreePath });
    return worktreePath;
  }

  if (lsRemote.exitCode === 2) {
    // Branch absent — standalone git init in temp dir (cannot use worktree add on non-existent ref)
    fs.mkdirSync(worktreePath, { recursive: true });
    await getExecOutput('git', ['init'], { cwd: worktreePath });
    await getExecOutput('git', ['remote', 'add', 'origin', repoRemoteUrl], { cwd: worktreePath });
    await getExecOutput('git', [
      '-c', `user.email=${BOT_EMAIL}`, '-c', `user.name=${BOT_NAME}`,
      'checkout', '--orphan', STATE_BRANCH
    ], { cwd: worktreePath });

    const today = todayPath();
    fs.mkdirSync(path.join(worktreePath, path.dirname(today)), { recursive: true });
    fs.writeFileSync(path.join(worktreePath, today), '', 'utf8');

    await getExecOutput('git', ['add', '-A'], { cwd: worktreePath });
    await getExecOutput('git', [
      '-c', `user.email=${BOT_EMAIL}`, '-c', `user.name=${BOT_NAME}`,
      'commit', '-m', `chore: init playwright-healer-state [skip-healer]`
    ], { cwd: worktreePath });

    const push = await getExecOutput(
      'git', ['push', '-u', 'origin', STATE_BRANCH],
      { cwd: worktreePath, ignoreReturnCode: true }
    );

    if (push.exitCode !== 0) {
      // Concurrent bootstrapper won — clean up and retry (branch now exists)
      fs.rmSync(worktreePath, { recursive: true, force: true });
      return bootstrapOrGetWorktree(repoRemoteUrl, primaryCwd);
    }

    return worktreePath;
  }

  throw new Error(
    `git ls-remote failed with exit code ${lsRemote.exitCode}: ${lsRemote.stderr}`
  );
}

export async function appendRecord(
  record: NdjsonRecord,
  worktreePath: string,
): Promise<void> {
  const ndjsonPath = todayPath();
  const absPath = path.join(worktreePath, ndjsonPath);

  for (let attempt = 0; attempt < MAX_RETRIES; attempt++) {
    // 1. Fetch + reset to remote state
    await getExecOutput('git', ['fetch', 'origin', STATE_BRANCH], { cwd: worktreePath });
    await getExecOutput('git', ['reset', '--hard', `origin/${STATE_BRANCH}`], { cwd: worktreePath });

    // 2. Ensure directory exists
    fs.mkdirSync(path.join(worktreePath, path.dirname(ndjsonPath)), { recursive: true });

    // 3. Atomic write (temp file rename prevents partial-write corruption — Pitfall B)
    const existing = fs.existsSync(absPath) ? fs.readFileSync(absPath, 'utf8') : '';
    const appended = existing + JSON.stringify(record) + '\n';
    const tmpPath = `${absPath}.tmp`;
    fs.writeFileSync(tmpPath, appended, 'utf8');
    fs.renameSync(tmpPath, absPath);

    // 4. Commit in worktree
    await getExecOutput('git', ['add', ndjsonPath], { cwd: worktreePath });
    await getExecOutput('git', [
      '-c', `user.email=${BOT_EMAIL}`, '-c', `user.name=${BOT_NAME}`,
      'commit', '-m', `stats: run ${record.runId} [skip-healer]`
    ], { cwd: worktreePath });

    // 5. Push with ref-qualified lease (Pitfall C — prevents stale FETCH_HEAD)
    const push = await getExecOutput('git', [
      'push',
      `--force-with-lease=${STATE_BRANCH}`,
      'origin',
      STATE_BRANCH,
    ], { cwd: worktreePath, ignoreReturnCode: true });

    if (push.exitCode === 0) return;

    // Rejected — exponential backoff + jitter
    const delayMs = 100 * Math.pow(2, attempt) + Math.floor(Math.random() * 100);
    core.warning(
      `State branch push rejected (attempt ${attempt + 1}/${MAX_RETRIES}). Retry in ${delayMs}ms.`
    );
    await new Promise((r) => setTimeout(r, delayMs));

    // Reset local commit before retry (fetch+reset at top of loop will re-sync)
    await getExecOutput('git', ['reset', '--soft', 'HEAD~1'], { cwd: worktreePath });
  }

  // Exhausted retries — non-fatal (Assumption A1 from RESEARCH.md)
  core.warning(
    `State branch: all ${MAX_RETRIES} push attempts rejected. ` +
    `Run ${record.runId} stats will not be recorded this time.`
  );
}

export async function removeWorktree(worktreePath: string): Promise<void> {
  // Pitfall J fix: ignoreReturnCode: true because temp-dir paths from standalone git init
  // are never registered as worktrees via 'git worktree add'
  await getExecOutput('git', ['worktree', 'remove', '--force', worktreePath], {
    ignoreReturnCode: true,
  });
  fs.rmSync(worktreePath, { recursive: true, force: true });
}
```

**Implement `runGc`** using the Pattern 4 date-walk logic:
1. Compute cutoff = new Date() - retentionDays days
2. Walk `runs/YYYY/MM/DD.ndjson` structure
3. Delete files where fileDate < cutoff
4. Remove empty month/year directories
5. If any files deleted: `git add -A` + `git commit -m "chore: gc healer state (retention Xd) [skip-healer]"`
6. Note: GC commit is pushed as part of the next appendRecord push; GC itself does not push separately
