  runId: string;          // GITHUB_RUN_ID

```typescript
// src/ingest/report-parser.ts
import { z } from 'zod';
import * as core from '@actions/core';
import type { NdjsonTestEntry } from '../shared/types.js';

const ReportSchema = z.object({
  config: z.object({ rootDir: z.string() }).optional(),
  suites: z.array(z.any()),  // deep validation done by walkSuites
  stats: z.object({ startTime: z.string(), duration: z.number() }).optional(),
});

export function parseReport(
  rawJson: unknown,
): { entries: NdjsonTestEntry[]; reportUnreadable: boolean } {
  const parsed = ReportSchema.safeParse(rawJson);
  if (!parsed.success) {
    core.warning(
      `ING-03: Playwright report does not match expected shape — ` +
      `recording as "report-unreadable". Zod issues: ${parsed.error.message}`
    );
    return { entries: [], reportUnreadable: true };
  }
  const entries: NdjsonTestEntry[] = [];
  walkSuites(parsed.data.suites, '', entries);
  return { entries, reportUnreadable: false };
}
```

IMPORTANT for testId construction: `testId = "{spec.file}::{fullTitle}"` where `fullTitle` = `${suiteTitle ? suiteTitle + ' > ' : ''}${spec.title}`. When there is no parent suite title (top-level spec), testId = `{spec.file}::{spec.title}`.

IMPORTANT for `test.status` mapping: Playwright test.status values are `'expected' | 'unexpected' | 'flaky' | 'skipped'` — NOT `'passed' | 'failed'`. The function `mapOutcome()` converts:
- `'expected'` → `'passed'`
- `'unexpected'` → `'failed'`
- `'flaky'` → `'flaky'`
- `'skipped'` → `'skipped'`
- any other value → `'failed'` (defensive fallback)

`retryCount` = `Math.max(0, ...(test.results ?? []).map((r: any) => r.retry ?? 0))`

`traceAttachmentPath` = `lastResult?.attachments?.find((a: any) => a.name === 'trace')?.path ?? null`

`errorSignature` = `lastResult?.error?.message?.slice(0, 200) ?? null`

**What NOT to do:**
- Do NOT use `result.status` ('passed'/'failed') for the outcome field — use `test.status` ('expected'/'unexpected') which is where Playwright computes 'flaky'.
- Do NOT validate nested suite/spec/test structure with strict Zod — use `z.array(z.any())` and manual walk.

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| GitHub Actions payload → loop guard | github.context.payload is provided by the runner; not user-controllable (fork repo cannot forge push payload author email) |
| Playwright JSON report → report-parser | Report file is read from GITHUB_WORKSPACE — on fork PRs, this is the fork's file; loop guard exits before report-parser is called |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-2-02 | Repudiation (loop amplification) | `src/shared/loop-guard.ts` — SEC-05 Guard 1 + Guard 2 | mitigate | Guard 1: bot email check on `payload.head_commit.author.email` prevents healer commits from triggering ingest. Guard 2: `[skip-healer]` sentinel in commit message provides an explicit escape hatch. Both use optional chaining to avoid crashes on non-push events (Pitfall D). Guard 0 exits on fork PRs before any write operations. |
| T-2-03 | Tampering (path traversal in report glob) | `src/ingest/report-parser.ts` — ING-01 glob resolution | mitigate | `@actions/glob` is used in 02-05 to resolve report-path, not in the parser itself. The parser receives the already-resolved file path from the glob result; it does not perform its own glob expansion. Glob resolution in 02-05 is constrained to GITHUB_WORKSPACE by @actions/glob semantics. |
| T-2-02b | Tampering (NDJSON injection via test title) | `src/ingest/report-parser.ts` — `JSON.stringify` in NDJSON record | accept | JSON.stringify() encodes all control characters including `\n`, `\r`. A test title containing newlines cannot inject a second NDJSON line. Risk: none. |
</threat_model>
